## 1. For EXP-ETCH-B, how much did the daily mean CD change between the first and last week of data?

Assessment: **NEEDS_REVIEW**

Genie SQL:
```sql
WITH bounds AS (
  SELECT MIN(capture_date) AS min_date, MAX(capture_date) AS max_date
  FROM ws_azure_classic_eastus_pedro.fabs_metrology.v_daily_cd_trend
  WHERE experiment_id = 'EXP-ETCH-B'
),
first_week AS (
  SELECT ROUND(try_divide(SUM(mean_cd_nm * images),SUM(images)), 2) AS first_week_mean_cd_nm,
         SUM(images) AS first_week_images
  FROM ws_azure_classic_eastus_pedro.fabs_metrology.v_daily_cd_trend t
  CROSS JOIN bounds b
  WHERE t.experiment_id = 'EXP-ETCH-B'
    AND t.mean_cd_nm IS NOT NULL
    AND t.capture_date >= b.min_date
    AND t.capture_date < DATE_ADD(b.min_date, 7)
),
last_week AS (
  SELECT ROUND(try_divide(SUM(mean_cd_nm * images),SUM(images)), 2) AS last_week_mean_cd_nm,
         SUM(images) AS last_week_images
  FROM ws_azure_classic_eastus_pedro.fabs_metrology.v_daily_cd_trend t
  CROSS JOIN bounds b
  WHERE t.experiment_id = 'EXP-ETCH-B'
    AND t.mean_cd_nm IS NOT NULL
    AND t.capture_date > DATE_SUB(b.max_date, 7)
    AND t.capture_date <= b.max_date
)
SELECT 'EXP-ETCH-B' AS experiment_id,
       first_week_mean_cd_nm,
       last_week_mean_cd_nm,
       ROUND(last_week_mean_cd_nm - first_week_mean_cd_nm, 2) AS change_nm,
       first_week_images,
       last_week_images
FROM first_week
CROSS JOIN last_week
```

Benchmark (expected) SQL:
```sql
WITH d AS (
  SELECT capture_date, SUM(mean_cd_nm * images) / SUM(images) AS daily_cd, SUM(images) AS images
  FROM ws_azure_classic_eastus_pedro.fabs_metrology.v_daily_cd_trend WHERE experiment_id = 'EXP-ETCH-B' GROUP BY capture_date
), lim AS (SELECT MIN(capture_date) AS d0, MAX(capture_date) AS d1 FROM d)
SELECT ROUND(SUM(CASE WHEN capture_date <= date_add(d0, 6) THEN daily_cd * images END)
           / SUM(CASE WHEN capture_date <= date_add(d0, 6) THEN images END), 2) AS first_week_cd_nm,
       ROUND(SUM(CASE WHEN capture_date >= date_sub(d1, 6) THEN daily_cd * images END)
           / SUM(CASE WHEN capture_date >= date_sub(d1, 6) THEN images END), 2) AS last_week_cd_nm,
       ROUND(SUM(CASE WHEN capture_date >= date_sub(d1, 6) THEN daily_cd * images END)
           / SUM(CASE WHEN capture_date >= date_sub(d1, 6) THEN images END)
           - SUM(CASE WHEN capture_date <= date_add(d0, 6) THEN daily_cd * images END)
           / SUM(CASE WHEN capture_date <= date_add(d0, 6) THEN images END), 2) AS change_nm
FROM d CROSS JOIN lim
```
